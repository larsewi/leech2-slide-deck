\section{Overview}

\begin{frame}
\frametitle{What is leech2?}
leech2 tracks changes to tables and turns them into small patches that can be applied
to a database.
\begin{itemize}
\item Computes deltas and stores them in linked blocks
\item Consolidates multiple blocks into a patch
\item Converts patches into SQL statements
\end{itemize}
\bigskip
Written in Rust and ships with a C FFI library and a CLI tool.
\begin{block}{Documentation}
See \texttt{lch(1)} and \texttt{leech2.h(3)}.
\end{block}
\end{frame}

\section{Setup}

\begin{frame}[fragile]
\frametitle{Initializing a project}
The \texttt{lch init} command creates a work directory with a starter CSV
table and a config file telling leech2 how to interpret it.
\begin{verbatim}
$ lch init
Initialized ./.leech2
$ ls .leech2/
config.toml  products.csv
\end{verbatim}
\end{frame}

\begin{frame}
\frametitle{The config file}
The leech2 config file can contain:
\begin{itemize}
\item Configuration options
\item Table definitions
\item Paths to drop-in fragments
\end{itemize}
\bigskip
\begin{block}{Note}
Both JSON and TOML syntax are supported.
\end{block}
\end{frame}

\section{Blocks}

\begin{frame}[fragile]
\frametitle{Creating the first block}
The \texttt{lch block create} command:
\begin{itemize}
\item Creates a new block
\item Records the current state
\end{itemize}
\medskip
\begin{verbatim}
$ lch block create
3294a7ec4d6f4bdb38c05fd94f0a1a8c6432f786
$ ls .leech2/state/
3294a7ec4d6f4bdb38c05fd94f0a1a8c6432f786  HEAD  STATE
\end{verbatim}
\end{frame}

\begin{frame}
\frametitle{The block files}
Blocks are content-addressed; the name is the SHA-1 digest of the contents.
Each block contains:
\begin{itemize}
\item Reference to its predecessor
\item Creation timestamp
\item Payload
\end{itemize}
\medskip
\blockchain
\end{frame}

\begin{frame}[fragile]
\frametitle{The first block}
The first block always contains:
\begin{itemize}
\item Reference to the genesis block
\item Empty payload
\end{itemize}
\medskip
\begin{verbatim}
$ lch block show
Block:
  Parent: 0000000000000000000000000000000000000000
  Created: 2026-08-25 10:32:44 UTC
  Payload (0 tables)
\end{verbatim}
\end{frame}

\begin{frame}[fragile]
\frametitle{The \texttt{HEAD} file}
\texttt{HEAD} is a plain text file that points to the last block in the chain.
It's updated whenever a new block is added.
\begin{verbatim}
$ cat .leech2/state/HEAD
3294a7ec4d6f4bdb38c05fd94f0a1a8c6432f786
\end{verbatim}
\medskip
\blockchain
\end{frame}

\begin{frame}[fragile]
\frametitle{The \texttt{STATE} file}
\texttt{STATE} holds snapshots of the tables as of the most recent block.
\begin{verbatim}
$ lch state show
State (1 tables):
  'products' [id, name, price]
    (2) "Mouse", 34.5
    (3) "Monitor", 249.95
    (1) "Keyboard", 79.99
\end{verbatim}
\begin{block}{Note}
Primary key columns are shown in parentheses.
\end{block}
\end{frame}

\section{Patches}

\begin{frame}[fragile]
\frametitle{Creating a patch}
The \texttt{lch patch create} command consolidates the changes and stores them
in the \texttt{PATCH} file.
\begin{verbatim}
$ lch patch create
3294a7ec4d6f4bdb38c05fd94f0a1a8c6432f786
$ ls .leech2/state/
3294a7ec4d6f4bdb38c05fd94f0a1a8c6432f786  PATCH  STATS
HEAD                                      STATE
\end{verbatim}
\begin{block}{Note}
Performance statistics are tracked in the \texttt{STATS} file.
\end{block}
\end{frame}

\begin{frame}
\frametitle{The \texttt{PATCH} file}
The patches contain:
\begin{itemize}
\item Reference and timestamp of the block at the head of the chain
\item Number of consolidated blocks
\item Full table contents or deltas
\end{itemize}
\medskip
\begin{block}{Note}
A patch may contain both deltas and full table contents. For each table, leech2
uses whichever is smaller.
\end{block}
\end{frame}

\begin{frame}[fragile]
\frametitle{The first patch}
The first patch contains only full table contents, since there is no earlier
state to diff against. Nothing was consolidated, hence \texttt{Blocks: 0}.
\begin{verbatim}
$ lch patch show
Patch:
  Head: 3294a7ec4d6f4bdb38c05fd94f0a1a8c6432f786
  Created: 2026-08-25 10:32:44 UTC
  Blocks: 0
  States (1):
    'products' [id, name, price]
      (2) "Mouse", 34.5
      (3) "Monitor", 249.95
      (1) "Keyboard", 79.99
\end{verbatim}
\end{frame}

\begin{frame}[fragile]
\frametitle{Rendering the patch as SQL}
\texttt{lch patch sql} renders the patch as SQL statements that replicate the
changes in the database.
\begin{verbatim}
$ lch patch sql
TRUNCATE "products";
INSERT INTO "products" ("id", "name", "price")
VALUES (2, 'Mouse', 34.5);
INSERT INTO "products" ("id", "name", "price")
VALUES (3, 'Monitor', 249.95);
INSERT INTO "products" ("id", "name", "price")
VALUES (1, 'Keyboard', 79.99);
\end{verbatim}
\end{frame}

\begin{frame}[fragile]
\frametitle{Marking the patch as applied}
\texttt{lch patch applied} records the last successful patch in the
\texttt{REPORTED} file.
\begin{verbatim}
$ lch patch applied
3294a7ec4d6f4bdb38c05fd94f0a1a8c6432f786
$ cat .leech2/state/REPORTED
3294a7ec4d6f4bdb38c05fd94f0a1a8c6432f786
\end{verbatim}
\begin{block}{Note}
Next patch will contain the consolidated data from \texttt{REPORTED} (exclusive) to
\texttt{HEAD}.
\end{block}
\end{frame}

\begin{frame}[fragile]
\frametitle{Marking the patch as failed}
\texttt{lch patch failed} removes the \texttt{REPORTED} file, so the next patch
contains full table contents instead of deltas.
\begin{verbatim}
$ lch patch failed --dry-run
Would have removed './.leech2/state/REPORTED'
0000000000000000000000000000000000000000
\end{verbatim}
\bigskip
\begin{block}{Note}
\texttt{--dry-run} reports what would happen without changing anything.
\end{block}
\end{frame}

\section{Walkthrough}

\begin{frame}[fragile]
\frametitle{Recording an update}

$ cat .leech2/products.csv

1,Keyboard,79.99
2,Mouse,29.99
3,Monitor,249.95
\end{verbatim}
\end{frame}

\begin{frame}[fragile]
\frametitle{Applying the patch}
Execute the following statements to mirror the table on the database and mark
the patch as applied.
\begin{verbatim}
$ lch patch sql
DELETE FROM "products"
WHERE "id" = 3;
UPDATE "products"
SET "price" = 29.99
WHERE "id" = 2;
$ lch patch applied
c4b22f2680f917bff7a61ad9f5463b52b7ee0861
\end{verbatim}

\section{Consolidation}

\begin{frame}
\frametitle{Block consolidation}
Every block in \texttt{REPORTED..HEAD} is merged into a single patch.
\begin{itemize}
\item 15 merging rules
\item \texttt{insert} and \texttt{delete} carry a primary key and values
\item \texttt{update} additionally carries the previous value
\item Never more than two blocks in memory at a time
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{No overlapping key}
When a key appears in one of two subsequent blocks, there is no conflict. The
operation passes through to the result unchanged.
\medskip
\begin{table}
\centering
\begin{tabular}{clll}
Rule & Parent & Child & Result \\
\hline
1  &                 & \texttt{insert} & \texttt{insert} \\
2  &                 & \texttt{delete} & \texttt{delete} \\
3  &                 & \texttt{update} & \texttt{update} \\
4  & \texttt{insert} &                 & \texttt{insert} \\
8  & \texttt{delete} &                 & \texttt{delete} \\
12 & \texttt{update} &                 & \texttt{update}
\end{tabular}
\end{table}
\end{frame}

\begin{frame}
\frametitle{Impossible combinations}
Some rules exist solely to detect logical errors.
\medskip
\begin{table}
\centering
\begin{tabular}{clll}
Rule & Parent & Child & Result \\
\hline
5  & \texttt{insert} & \texttt{insert} & \texttt{error} \\
10 & \texttt{delete} & \texttt{delete} & \texttt{error} \\
11 & \texttt{delete} & \texttt{update} & \texttt{error} \\
13 & \texttt{update} & \texttt{insert} & \texttt{error}
\end{tabular}
\end{table}
\begin{example}
You cannot \texttt{insert} something that already exists (rules 5, 13),
\texttt{delete} something that is already gone (rule 10), or \texttt{update}
something that has been deleted (rule 11).
\end{example}
\end{frame}

\begin{frame}
\frametitle{\texttt{insert} then \texttt{delete}}
\begin{table}
\centering
\begin{tabular}{cllll}
Rule & Parent & Child & Condition & Result \\
\hline
6a & \texttt{insert} & \texttt{delete} & values match  &                \\
6b & \texttt{insert} & \texttt{delete} & values differ & \texttt{error} \\
\end{tabular}
\end{table}
\begin{example}
An \texttt{insert} followed by a \texttt{delete} cancels out -- the net effect
is the same as never inserting the row. However, if the deleted value differs
from the inserted one, then this is a logical error.
\end{example}
\end{frame}

\begin{frame}
\frametitle{\texttt{insert} then \texttt{update}}
\begin{table}
\centering
\begin{tabular}{cllll}
Rule & Parent & Child & Condition & Result \\
\hline
7a & \texttt{insert} & \texttt{update} & values match  & \texttt{insert} \\
7b & \texttt{insert} & \texttt{update} & values differ & \texttt{error}  \\
\end{tabular}
\end{table}
\begin{example}
An \texttt{insert} followed by an \texttt{update} collapses into a single
\texttt{insert} carrying the updated value. However, if the inserted value differs
from the update's old value, then this is a logical error.
\end{example}
\end{frame}

\begin{frame}
\frametitle{\texttt{delete} then \texttt{insert}}
\begin{table}
\centering
\begin{tabular}{cllll}
Rule & Parent & Child & Condition & Result \\
\hline
9a & \texttt{delete} & \texttt{insert} & values match  &                 \\
9b & \texttt{delete} & \texttt{insert} & values differ & \texttt{update} \\
\end{tabular}
\end{table}
\begin{example}
A \texttt{delete} followed by an \texttt{insert} cancels out -- the net effect
is no change at all. However, if the inserted value differs from the deleted
one, the pair collapses into a single \texttt{update} from the deleted value to
the inserted one.
\end{example}
\end{frame}

\begin{frame}
\frametitle{\texttt{update} then \texttt{delete}}
\begin{table}
\centering
\begin{tabular}{cllll}
Rule & Parent & Child & Condition & Result \\
\hline
14a & \texttt{update} & \texttt{delete} & values match  & \texttt{delete} \\
14b & \texttt{update} & \texttt{delete} & values differ & \texttt{error}  \\
\end{tabular}
\end{table}
\begin{example}
An \texttt{update} followed by a \texttt{delete} collapses into a single
\texttt{delete} carrying the update's old value. But if the deleted value
differs from the update's new value, then this is a logical error.
\end{example}
\end{frame}

\begin{frame}
\frametitle{\texttt{update} then \texttt{update}}
\begin{table}
\centering
\begin{tabular}{cllll}
Rule & Parent & Child & Condition & Result \\
\hline
15a & \texttt{update} & \texttt{update} & net change    & \texttt{update} \\
15b & \texttt{update} & \texttt{update} & no net change &                 \\
15c & \texttt{update} & \texttt{update} & values differ & \texttt{error}  \\
\end{tabular}
\end{table}
\begin{example}
Two stacked \texttt{update} operations collapse into a single \texttt{update}
from the parent's old value to the child's new value. If those two values are
equal, the pair cancels out. However, if intermediate values differ, then this
is a logical error.
\end{example}
\end{frame}
